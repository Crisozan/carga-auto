        // ANALIZAR ETIQUETA CON GEMINI API
        async function processImageWithAI() {
            const apiKey = localStorage.getItem('gemini_api_key');
            if (!apiKey) {
                alert('Por favor configura tu API Key de Gemini desde el botón superior.');
                openApiKeyModal();
                return;
            }

            if (!currentImageData) return;

            const btn = document.getElementById('btnProcessAI');
            btn.disabled = true;
            btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Procesando Etiqueta...';

            try {
                const base64Data = currentImageData.split(',')[1];
                const promptText = `Analiza esta imagen de una etiqueta de producción industrial de bolsas plásticas. Extrae los siguientes campos en formato JSON estricto:
                - provider: Texto del proveedor en la parte superior (ej: "OXI")
                - providerDate: Fecha escrita al lado del proveedor (ej: "07/10")
                - specCode: Código de especificación en texto impreso (ej: "60 R" o "60RB", devuélvelo sin espacios intermedios)
                - weight: Número decimal del PESO en KG (ej: 542)
                - rolls: Número de la CANTIDAD de rollos (ej: 1800)
                - batch: Número del LOTE (ej: "087")
                - prodDate: Fecha de producción en formato AAAA-MM-DD si es visible, de lo contrario deja cadena vacía.
                Responde ÚNICAMENTE con el objeto JSON sintácticamente válido.`;

                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-1.5-flash:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{
                            parts: [
                                { text: promptText },
                                { inline_data: { mime_type: "image/jpeg", data: base64Data } }
                            ]
                        }]
                    })
                });

                const data = await response.json();
                const rawText = data.candidates[0].content.parts[0].text;
                const cleanJson = rawText.replace(/```json/g, '').replace(/```/g, '').trim();
                const parsed = JSON.parse(cleanJson);

                // Autorellenar formulario
                if (parsed.provider) document.getElementById('formProvider').value = parsed.provider;
                if (parsed.providerDate) document.getElementById('formProviderDate').value = parsed.providerDate;
                if (parsed.specCode) document.getElementById('formSpecCode').value = parsed.specCode.replace(/\s+/g, '');
                if (parsed.weight) document.getElementById('formWeight').value = parsed.weight;
                if (parsed.rolls) document.getElementById('formRolls').value = parsed.rolls;
                if (parsed.batch) document.getElementById('formBatch').value = parsed.batch;
                if (parsed.prodDate) document.getElementById('formProdDate').value = parsed.prodDate;

                document.getElementById('aiBadge').classList.remove('hidden');
                updateSpecMatchNotice();

            } catch (err) {
                console.error(err);
                alert('No se pudo analizar la etiqueta. Verifica tu API Key o revisa los datos manualmente.');
            } finally {
                btn.disabled = false;
                btn.innerHTML = '<i class="fa-solid fa-wand-magic-sparkles"></i> Analizar con IA (Gemini)';
            }
        }

        // VISTA AGRUPADA POR ESPECIFICACIÓN
        function renderGroupedView() {
            const container = document.getElementById('groupedContainer');
            const filter = document.getElementById('groupDateFilter').value;
            container.innerHTML = '';

            const todayStr = new Date().toISOString().split('T')[0];
            const currentMonthStr = todayStr.substring(0, 7);

            // Filtrar registros
            let filteredRecords = records.filter(r => {
                if (filter === 'today') return r.prodDate === todayStr;
                if (filter === 'month') return r.prodDate.startsWith(currentMonthStr);
                return true;
            });

            // Agrupar por specCode
            const groups = {};
            filteredRecords.forEach(r => {
                if (!groups[r.specCode]) {
                    groups[r.specCode] = {
                        specCode: r.specCode,
                        totalWeight: 0,
                        totalRolls: 0,
                        items: []
                    };
                }
                groups[r.specCode].totalWeight += r.weight;
                groups[r.specCode].totalRolls += r.rolls;
                groups[r.specCode].items.push(r);
            });

            const keys = Object.keys(groups);
            document.getElementById('groupedTotalCount').innerText = keys.length;

            if (keys.length === 0) {
                container.innerHTML = `<div class="col-span-full text-center py-12 text-slate-500">No hay registros para mostrar en el período seleccionado.</div>`;
                return;
            }

            keys.forEach(code => {
                const g = groups[code];
                const catItem = catalog[code] || { description: 'Sin catálogo asignado', color: 'Otro' };
                const avgKg = (g.totalWeight / g.items.length).toFixed(1);

                const cardHtml = `
                    <div class="bg-[#111c38] border border-[#1e2d54] rounded-xl overflow-hidden flex flex-col justify-between shadow-md">
                        <div class="p-4 space-y-3">
                            <div class="flex justify-between items-start">

                                <div class="bg-[#0b1329] p-2 rounded-lg">
                                    <p class="text-[10px] text-slate-400">Total KG</p>
                                    <p class="text-sm font-bold text-cyan-400">${g.totalWeight} kg</p>
                                </div>
                                <div class="bg-[#0b1329] p-2 rounded-lg">
                                    <p class="text-[10px] text-slate-400">Total Rollos</p>

                                </div>

                        </div>

                            <i class="fa-solid fa-trash"></i>
                        </button>
                    </td>
                `;

            });

        function updateKPIs() {
            const totalW = records.reduce((acc, r) => acc + r.weight, 0);
            const totalR = records.reduce((acc, r) => acc + r.rolls, 0);
            const totalL = records.length;
            const avgW = totalL > 0 ? (totalW / totalL).toFixed(1) : 0;

            document.getElementById('kpiWeight').innerText = `${totalW} kg`;
            document.getElementById('kpiRolls').innerText = totalR;
            document.getElementById('kpiLots').innerText = totalL;
            document.getElementById('kpiAvgWeight').innerText = `${avgW} kg`;
        }

        function deleteRecord(id) {
            if (!confirm('¿Seguro que deseas eliminar este registro de producción?')) return;
            records = records.filter(r => r.id !== id);
            localStorage.setItem('prod_records', JSON.stringify(records));
            renderHistoryTable();
            updateKPIs();
        }
